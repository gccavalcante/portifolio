# Estrutura representativa da busca indexada (Q Objects)
# e automação de diretórios com Django Signals

from django.db.models import Q
from django.db.models.signals import post_save
from django.dispatch import receiver

def buscar_editais_indexados(termo: str, secretaria: str | None, status: str | None):
    filtro = Q(publicado=True)
    if termo:
        filtro &= (
            Q(numero_edital__icontains=termo)
            | Q(titulo__icontains=termo)
            | Q(objeto_resumo__icontains=termo)
        )
    if secretaria:
        filtro &= Q(secretaria__sigla__iexact=secretaria)
    if status:
        filtro &= Q(situacao=status)

    return (
        EditalPagina.objects.filter(filtro)
        .select_related("secretaria")
        .order_by("-data_publicacao")
    )

@receiver(post_save, sender=EditalPagina)
def organizar_diretorio_edital(sender, instance, created, **kwargs):
    # Garante árvore padronizada /ano/secretaria/numero_edital sem ação manual
    instance.sincronizar_estrutura_arquivos()

-- Gestão centralizada de clientes, serviços e contratos

CREATE TABLE IF NOT EXISTS clients (
  id VARCHAR(36) PRIMARY KEY,
  name VARCHAR(150) NOT NULL,
  document VARCHAR(20) NOT NULL UNIQUE,
  email VARCHAR(120) NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS contracts (
  id VARCHAR(36) PRIMARY KEY,
  client_id VARCHAR(36) NOT NULL,
  service_description VARCHAR(255) NOT NULL,
  status ENUM('ACTIVE', 'PENDING', 'CANCELED') NOT NULL DEFAULT 'PENDING',
  value DECIMAL(12, 2) NOT NULL,
  start_date DATE NOT NULL,
  end_date DATE NULL,
  CONSTRAINT fk_contract_client
    FOREIGN KEY (client_id) REFERENCES clients(id)
    ON DELETE RESTRICT ON UPDATE CASCADE
);

-- Consolidação de receita contratual e taxa de retenção por status
SELECT
  DATE_FORMAT(c.start_date, '%Y-%m') AS safra_mes,
  COUNT(c.id) AS total_contratos,
  SUM(CASE WHEN c.status = 'ACTIVE' THEN c.value ELSE 0 END) AS receita_ativa,
  ROUND(
    100.0 * SUM(CASE WHEN c.status = 'ACTIVE' THEN 1 ELSE 0 END) / COUNT(c.id),
    2
  ) AS taxa_ativacao_pct
FROM contracts c
INNER JOIN clients cl ON cl.id = c.client_id
GROUP BY DATE_FORMAT(c.start_date, '%Y-%m')
ORDER BY safra_mes DESC;
